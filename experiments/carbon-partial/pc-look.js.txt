// pc-look.js
// PC LOOK (2026-11-24, experiments/2026-11-24-pc-look-plan.md): Bot can list Omri's open Chrome tabs (title + address) and open PC
// windows (program + title), and look at ONE tab (its page text, or a picture of it) or ONE window (a picture of it).
//
// Read-only, never control: Bot never clicks, types, focuses, unminimizes, switches tabs, moves or closes windows.
// Rules enforced here in code (never left to a model):
//   - NEVER_ALLOWED: password managers, chrome:// / edge:// / about: / file:// pages, browser password pages, and banking sites/apps
//     are refused always — even if someone put one in data/pc-look.json or tapped a card. Adding one to a list is refused with a plain reason.
//   - Targets come ONLY by NUMBER from the last list Bot read (kept 2 hours, like pc-find.js); a model never hands in an address or handle.
//   - Two allowed lists in data/pc-look.json: sites (e.g. "moodle.bgu.ac.il") and apps (e.g. "Code.exe"). Looking at an item on its list
//     runs at once; anything else returns a single-use 15-minute Confirm card ("Just this once", "Always allow (add to the list)", "Cancel").
//   - The lists change ONLY when Omri wrote the name in his own message (checked by siteInText / appInText) or tapped the card/button.
//   - Limits: at most 3 looks per message (checked in brain.js) and 30 per hour (checked here); pictures at most 8 MB, real JPEG/PNG first bytes.
//   - Every real look is logged (what was looked at, never its content); temporary screenshot files get random names and are deleted after use.

// Friendly spoken names that map to exact Windows program names ONLY when Omri wrote that word in his own message.
const KNOWN_APP_ALIASES = {
  "code.exe": ["code", "vs code", "vscode"],
  "systemsettings.exe": ["systemsettings", "settings"],

const fail = (message, extra = {}) => Object.assign(new Error(message), { code: "refused", ...extra });
const oneLine = (s, n = 180) => String(s == null ? "" : s).replace(/[\u0000-\u001f\u007f]+/g, " ").replace(/\s+/g, " ").trim().slice(0, n);
const shortApp = (app) => String(app || "").replace(/\.exe$/i, "");

// Real JPEG or PNG first bytes only (spec: "a picture at most 8 MB, real JPEG/PNG first bytes checked").
function sniffLookImage(bytes) {
  if (!bytes || bytes.length < 4) return null;
  if (bytes[0] === 0xff && bytes[1] === 0xd8 && bytes[2] === 0xff) return ".jpg";
  if (bytes.length >= 8 && Buffer.from(bytes.subarray(0, 8)).equals(Buffer.from([0x89, 0x50, 0x4e, 0x47, 0x0d, 0x0a, 0x1a, 0x0a]))) return ".png";
  return null;
}

function checkLookImage(raw) {
  const bytes = Buffer.isBuffer(raw) ? raw : raw instanceof Uint8Array ? Buffer.from(raw) : null;
  if (!bytes || !bytes.length) throw fail("the picture came back empty");
  if (bytes.length > MAX_LOOK_IMAGE_BYTES) throw fail("the picture is bigger than 8 MB, so Bot refused it");
  const ext = sniffLookImage(bytes);
  if (!ext) throw fail("the picture is not a real JPEG or PNG image (checked by its first bytes)");
  return { bytes, ext };
}

function siteOfUrl(rawUrl) {
  let u;
  try { u = new URL(String(rawUrl || "").trim()); } catch (error) { return null; }
  if (u.protocol !== "https:" && u.protocol !== "http:") return null;
  return u.hostname.toLowerCase().replace(/\.$/, "");
}

function whyNeverSite(host) {
  const h = String(host || "").toLowerCase().replace(/\.$/, "");
  if (!h) return "that is not a valid site name";
  for (const s of NEVER_ALLOWED.sites) {
    if (h === s || h.endsWith("." + s)) {
      const kind = /password|bitwarden|keepass|lastpass|dashlane|nordpass|keeper|proton/.test(s) ? "password managers" : "banking and payment sites";
      return `Bot never looks at ${kind} (${s}) — they are blocked even on the allowed list`;
    }
  }
  if (NEVER_ALLOWED.hostPattern.test(h)) {
    const kind = /password|bitwarden|keepass|lastpass|dashlane|nordpass|enpass|roboform/i.test(h) ? "password managers" : "banking sites";
    return `Bot never looks at ${kind} (${h}) — they are blocked even on the allowed list`;
  }
  return null;
}

function whyNeverTab(rawUrl, title = "") {
  const raw = String(rawUrl || "").trim();
  if (!raw) return "that tab has no readable web address (it may be an internal browser page)";
  const schemeMatch = /^([a-z][a-z0-9+.-]*:)/i.exec(raw);
  const scheme = schemeMatch ? schemeMatch[1].toLowerCase() : "";
  if (NEVER_ALLOWED.schemes.includes(scheme)) {
    if (/password/i.test(raw) || /logins/i.test(raw)) return `Bot never looks at the browser's password page (${scheme})`;
    return `Bot never looks at ${scheme}// pages (internal browser and local file pages are never allowed)`.replace(":///", "://").replace("about://", "about:");
  }
  let u;
  try { u = new URL(raw); } catch (error) { return "that tab does not have a standard http/https address"; }
  if (u.protocol !== "https:" && u.protocol !== "http:") return `Bot never looks at ${u.protocol} pages`;
  const bySite = whyNeverSite(u.hostname);
  if (bySite) return bySite;
  if (NEVER_ALLOWED.titlePattern.test(String(title || ""))) return "Bot never looks at password managers, browser password pages or banking pages";
  return null;
}

function whyNeverApp(app) {
  const stem = shortApp(path.basename(String(app || "").trim())).toLowerCase();
  if (!stem) return "which app?";
  if (NEVER_ALLOWED.apps.includes(stem) || /1password|bitwarden|keepass|lastpass|dashlane|nordpass|enpass|roboform|keeper/i.test(stem)) {
    return `Bot never looks at password managers (${oneLine(app, 60)}) — they are blocked even on the allowed list`;
  }
  if (/^(?:bank|netbank)/i.test(stem)) {
    return `Bot never looks at banking apps (${oneLine(app, 60)}) — they are blocked even on the allowed list`;
  }
  return null;
}

function whyNeverWindow(app, title = "") {
  const byApp = whyNeverApp(app);
  if (byApp) return byApp;
  if (NEVER_ALLOWED.titlePattern.test(String(title || ""))) {
    return "that window shows a password manager, browser internal page or banking page, which Bot never looks at";
  }
  return null;
}

function normalizeSite(raw) {
  const text = String(raw || "").trim();
  if (!text) throw fail("give a site name like moodle.bgu.ac.il");
  if (/[\s\u0000-\u001f\u007f]/.test(text)) throw fail("that site name holds spaces or control characters");
  const schemeMatch = /^([a-z][a-z0-9+.-]*:)/i.exec(text);
  if (schemeMatch) {
    const why = whyNeverTab(text, "");
    if (why) throw fail(why);
    const host = siteOfUrl(text);
    if (!host) throw fail("only http/https sites like moodle.bgu.ac.il can be allowed");
    const bad = whyNeverSite(host);
    if (bad) throw fail(bad);
    return host;
  }
  const clean = text.replace(/\/.*$/, "").toLowerCase().replace(/\.$/, "");
  if (!/^[a-z0-9](?:[a-z0-9-]{0,61}[a-z0-9])?(?:\.[a-z0-9](?:[a-z0-9-]{0,61}[a-z0-9])?)*$/.test(clean) || (clean !== "localhost" && !clean.includes("."))) {
    throw fail(`"${oneLine(raw, 80)}" is not a valid site host (for example moodle.bgu.ac.il)`);
  }
  const bad = whyNeverSite(clean);
  if (bad) throw fail(bad);
  return clean;
}

function normalizeApp(raw) {
  const text = String(raw || "").trim().replace(/^["']|["']$/g, "");
  if (!text) throw fail("give a program name like Code.exe or Spotify.exe");
  if (/[/\\\u0000-\u001f\u007f]/.test(text)) throw fail("give only the program name (like Code.exe), not a folder path");
  const compact = text.toLowerCase();
  for (const [canonical, names] of Object.entries(KNOWN_APP_ALIASES)) {
    if (names.includes(compact)) {
      const display = canonical === "code.exe" ? "Code.exe" : canonical === "systemsettings.exe" ? "SystemSettings.exe" : "msedge.exe";
      const bad = whyNeverApp(display);
      if (bad) throw fail(bad);
      return display;
    }
  }
  const stem = shortApp(text);
  if (!/^[A-Za-z0-9._-]{2,60}$/.test(stem)) throw fail(`"${oneLine(raw, 60)}" is not a valid program name (for example Spotify.exe)`);
  const out = `${stem}.exe`;
  const bad = whyNeverApp(out);
  if (bad) throw fail(bad);
  return out;
}

const escRe = (s) => String(s).replace(/[.*+?^${}()|[\]\\]/g, "\\$&");

// Did Omri really write this site in his own message? Prevents a model from adding a domain Omri never named.
function siteInText(task, site) {
  const hay = String(task || "").toLowerCase();
  const want = String(site || "").toLowerCase();
  if (!want) return false;
  const re = new RegExp(`(^|[^a-z0-9.-])${escRe(want)}($|[^a-z0-9.-])`, "i");
  return re.test(hay);
}

// Did Omri really write this app name (with or without .exe, or a known alias like "VS Code" / "Settings") in his own message?
function appInText(task, app) {
  const hay = String(task || "").toLowerCase();
  const low = String(app || "").toLowerCase();
  const stem = shortApp(low);
  if (stem.length < 2) return false;
  const candidates = [low, stem, ...(KNOWN_APP_ALIASES[low] || [])];
  return candidates.some((w) => new RegExp(`(^|[^a-z0-9._-])${escRe(w)}($|[^a-z0-9._-])`, "i").test(hay));
}

function parseWindowsJson(stdout) {
  const raw = String(stdout || "").trim();
  if (!raw) return [];
  let parsed = JSON.parse(raw);
  if (Array.isArray(parsed) && parsed.length === 1 && Array.isArray(parsed[0])) parsed = parsed[0];
  const arr = Array.isArray(parsed) ? parsed : parsed && typeof parsed === "object" ? [parsed] : [];
  const out = [];
  for (const w of arr.slice(0, MAX_WINDOWS)) {
    if (!w || typeof w !== "object") continue;
    const hwnd = String(w.hwnd == null ? "" : w.hwnd).trim();
    if (!/^-?\d{1,20}$/.test(hwnd) || hwnd === "0") continue;
    const rawApp = oneLine(w.app, 80);
    const app = rawApp ? (/\.exe$/i.test(rawApp) ? rawApp : `${rawApp}.exe`) : "unknown.exe";
    const title = oneLine(w.title, 240);
    if (!title) continue;
    out.push({ hwnd, app, title, minimized: Boolean(w.minimized) });
  }
  return out;
}

function runPsScript(execFile, script, extraEnv = {}) {
  return new Promise((resolve, reject) => {
    execFile(
      "powershell.exe",
      ["-NoProfile", "-NonInteractive", "-Command", script],
      { windowsHide: true, timeout: 15000, maxBuffer: 10 * 1024 * 1024, env: { ...process.env, ...extraEnv } },
      (error, stdout, stderr) => {
        if (error) return reject(new Error(oneLine(stderr || error.message, 200) || "PowerShell failed"));
        resolve(String(stdout || ""));
      }
    );
  });
}

function createPcLook({
  dataDir,
  bridge = null,
  listWindows = null,
  captureWindow = null,
  execFile = realExecFile,
  tempRoot = path.join(os.tmpdir(), "bot-pclook"),
  log = console.log,
  now = () => Date.now(),
}) {
  const storeFile = path.join(dataDir, "pc-look.json");
  let lastTabs = { at: 0, items: [] };
  let lastWindows = { at: 0, items: [] };
  const pending = new Map(); // cardId -> pending look
  const hourLooks = []; // timestamps of executed looks in the last hour

  const pruneHour = () => {
    const cutoff = now() - HOUR_MS;
    while (hourLooks.length && hourLooks[0] <= cutoff) hourLooks.shift();
  };
  const sweepCards = () => {
    for (const [id, p] of pending) if (now() - p.at > CARD_EXPIRES_MS) pending.delete(id);
  };
  const cleanStaleTemp = () => {
    try {
      for (const f of fs.readdirSync(tempRoot)) {
        const full = path.join(tempRoot, f);
        try { if (now() - fs.statSync(full).mtimeMs > HOUR_MS) fs.rmSync(full, { recursive: true, force: true }); } catch (error) { /* in use */ }
      }
    } catch (error) { /* no folder yet */ }
  };

  function readStore() {
    try {
      const j = JSON.parse(fs.readFileSync(storeFile, "utf8"));
      const sites = Array.isArray(j && j.sites) ? j.sites.filter((x) => typeof x === "string" && !whyNeverSite(x)).map((x) => x.toLowerCase()) : [];
      const apps = Array.isArray(j && j.apps) ? j.apps.filter((x) => typeof x === "string" && !whyNeverApp(x)) : [];
      return { sites: [...new Set(sites)], apps };
    } catch (error) {
      return { sites: [], apps: [] };
    }
  }

  function writeStore(next) {
    fs.mkdirSync(dataDir, { recursive: true });
    const tmp = storeFile + ".tmp";
    fs.writeFileSync(tmp, JSON.stringify({ sites: next.sites, apps: next.apps }, null, 2));
    fs.renameSync(tmp, storeFile);
  }

  const isSiteAllowed = (site) => {
    if (!site || whyNeverSite(site)) return false;
    const low = site.toLowerCase();
    return readStore().sites.some((s) => s === low || low === "www." + s || "www." + low === s);
  };

  const isAppAllowed = (app) => {
    if (!app || whyNeverApp(app)) return false;
    const want = shortApp(app).toLowerCase();
    return readStore().apps.some((a) => shortApp(a).toLowerCase() === want);
  };

  function lists() {
    return readStore();
  }

  function listsText() {
    const { sites, apps } = readStore();
    return `Allowed sites: ${sites.length ? sites.join(", ") : "(none)"}.\nAllowed apps: ${apps.length ? apps.join(", ") : "(none)"}.`;
  }

  function allow({ site, app, remove = false, task = "", fromUi = false } = {}) {
    const hasSite = site !== undefined && site !== null && String(site).trim() !== "";
    const hasApp = app !== undefined && app !== null && String(app).trim() !== "";
    if (hasSite && hasApp) throw fail('give either "site" or "app" in one call, not both');
    if (!hasSite && !hasApp) throw fail('give "site" (like moodle.bgu.ac.il) or "app" (like Spotify.exe)');
    const cur = readStore();

    if (hasSite) {
      if (remove) {
        const want = String(site).trim().replace(/^https?:\/\//i, "").replace(/\/.*$/, "").toLowerCase();
        const nextSites = cur.sites.filter((s) => s.toLowerCase() !== want);
        if (nextSites.length === cur.sites.length) return { changed: false, kind: "site", value: want, ...cur, text: `"${oneLine(want, 80)}" was not on the allowed sites list.` };
        writeStore({ sites: nextSites, apps: cur.apps });
        log(`[pc-look] removed allowed site: ${want}`);
        return { changed: true, kind: "site", value: want, sites: nextSites, apps: cur.apps, text: `Removed "${want}" from the allowed sites list.` };
      }
      const clean = normalizeSite(site);
      if (!fromUi && !siteInText(task, clean)) {
        throw fail(`the site "${clean}" must be written in Omri's own message (or confirmed on a card)`);
      }
      if (cur.sites.includes(clean)) return { changed: false, kind: "site", value: clean, ...cur, text: `"${clean}" is already on the allowed sites list.` };
      const nextSites = [...cur.sites, clean];
      writeStore({ sites: nextSites, apps: cur.apps });
      log(`[pc-look] added allowed site: ${clean}`);
      return { changed: true, kind: "site", value: clean, sites: nextSites, apps: cur.apps, text: `Added "${clean}" to the allowed sites list. Bot can now look at tabs on ${clean} without asking.` };
    }

    if (remove) {
      const wantStem = shortApp(String(app).trim()).toLowerCase();
      const removed = cur.apps.find((a) => shortApp(a).toLowerCase() === wantStem);
      if (!removed) return { changed: false, kind: "app", value: oneLine(app, 60), ...cur, text: `"${oneLine(app, 60)}" was not on the allowed apps list.` };
      const nextApps = cur.apps.filter((a) => shortApp(a).toLowerCase() !== wantStem);
      writeStore({ sites: cur.sites, apps: nextApps });
      log(`[pc-look] removed allowed app: ${removed}`);
      return { changed: true, kind: "app", value: removed, sites: cur.sites, apps: nextApps, text: `Removed "${removed}" from the allowed apps list.` };
    }
    const clean = normalizeApp(app);
    if (!fromUi && !appInText(task, clean)) {
      throw fail(`the app "${clean}" must be written in Omri's own message (or confirmed on a card)`);
    }
    if (cur.apps.some((a) => shortApp(a).toLowerCase() === shortApp(clean).toLowerCase())) {
      return { changed: false, kind: "app", value: clean, ...cur, text: `"${clean}" is already on the allowed apps list.` };
    }
    const nextApps = [...cur.apps, clean];
    writeStore({ sites: cur.sites, apps: nextApps });
    log(`[pc-look] added allowed app: ${clean}`);
    return { changed: true, kind: "app", value: clean, sites: cur.sites, apps: nextApps, text: `Added "${clean}" to the allowed apps list. Bot can now look at ${clean} windows without asking.` };
  }

  const defaultListWindows = async () => parseWindowsJson(await runPsScript(execFile, LIST_WINDOWS_PS));
  const defaultCaptureWindow = async (hwnd) => {
    if (!/^-?\d{1,20}$/.test(String(hwnd)) || String(hwnd) === "0") throw fail("invalid window handle");
    fs.mkdirSync(tempRoot, { recursive: true });
    const rawFile = path.join(tempRoot, "cap-" + crypto.randomBytes(12).toString("hex") + ".img");
    try {
      const out = await runPsScript(execFile, CAPTURE_WINDOW_PS, { BOT_LOOK_HWND: String(hwnd), BOT_LOOK_OUT: rawFile });
      const info = JSON.parse(String(out || "{}").trim() || "{}");
      if (info.status !== "ok") return { status: info.status || "failed", title: oneLine(info.title, 240) };
      const bytes = fs.readFileSync(rawFile);
      return { status: "ok", title: oneLine(info.title, 240), bytes };
    } finally {
      try { fs.rmSync(rawFile, { force: true }); } catch (error) { /* already gone */ }
    }
  };

  const getWindows = listWindows || defaultListWindows;
  const getWindowShot = captureWindow || defaultCaptureWindow;

  function saveLookPhoto(checked) {
    cleanStaleTemp();
    fs.mkdirSync(tempRoot, { recursive: true });
    const file = path.join(tempRoot, crypto.randomBytes(16).toString("hex") + checked.ext);
    fs.writeFileSync(file, checked.bytes, { flag: "wx" });
    return { path: file, ext: checked.ext };
  }

  async function readTabs() {
    if (!bridge) throw fail("Chrome tab access is not set up in this Bot", { code: "offline" });
    const res = await bridge.ask("tabs", {});
    const raw = Array.isArray(res && res.data) ? res.data : [];
    const items = [];
    for (const t of raw.slice(0, MAX_TABS)) {
      if (!t || !Number.isInteger(Number(t.id)) || Number(t.id) <= 0) continue;
      const url = oneLine(t.url !== undefined ? t.url : t.address, 600);
      const title = oneLine(t.title, 240) || "(untitled)";
      items.push({
        n: items.length + 1,
        id: Number(t.id),
        title,
        url,
        site: siteOfUrl(url),
        active: Boolean(t.active),
        minimized: Boolean(t.minimized),
      });
    }
    lastTabs = { at: now(), items };
    const text = items.length
      ? items.map((t) => {
          const flags = [];
          if (whyNeverTab(t.url, t.title)) flags.push("never allowed");
          else if (isSiteAllowed(t.site)) flags.push("allowed");
          if (t.minimized) flags.push("window minimized");
          else if (!t.active) flags.push("background tab");
          return `${t.n}. ${t.title} — ${t.url || "(no address)"}${flags.length ? ` [${flags.join(", ")}]` : ""}`;
        }).join("\n")
      : "No open Chrome tabs were reported.";
    return { items, text };
  }

  async function readWindows() {
    const raw = await getWindows();
    const list = Array.isArray(raw) ? raw : [];
    const items = [];
    for (const w of list.slice(0, MAX_WINDOWS)) {
      if (!w || typeof w !== "object") continue;
      const hwnd = String(w.hwnd == null ? "" : w.hwnd).trim();
      if (!hwnd || hwnd === "0") continue;
      const rawApp = oneLine(w.app, 80) || "unknown.exe";
      const app = /\.exe$/i.test(rawApp) ? rawApp : `${rawApp}.exe`;
      const title = oneLine(w.title, 240);
      if (!title) continue;
      items.push({ n: items.length + 1, hwnd, app, title, minimized: Boolean(w.minimized) });
    }
    lastWindows = { at: now(), items };
    const text = items.length
      ? items.map((w) => {
          const flags = [];
          if (whyNeverWindow(w.app, w.title)) flags.push("never allowed");
          else if (isAppAllowed(w.app)) flags.push("allowed");
          if (w.minimized) flags.push("minimized");
          return `${w.n}. [${w.app}] ${w.title}${flags.length ? ` (${flags.join(", ")})` : ""}`;
        }).join("\n")
      : "No open windows were found on the PC.";
    return { items, text };
  }

  function pickTab(number) {
    const k = Number(number);
    if (!Number.isInteger(k) || k < 1) throw fail("say which tab number from the last chrome_tabs list");
    if (!lastTabs.items.length || now() - lastTabs.at > KEEP_MS) throw fail("there is no recent Chrome tab list; call chrome_tabs first");
    const hit = lastTabs.items.find((x) => x.n === k);
    if (!hit) throw fail(`the last Chrome tab list has only ${lastTabs.items.length} tab(s)`);
    return hit;
  }

  function pickWindow(number) {
    const k = Number(number);
    if (!Number.isInteger(k) || k < 1) throw fail("say which window number from the last pc_windows list");
    if (!lastWindows.items.length || now() - lastWindows.at > KEEP_MS) throw fail("there is no recent window list; call pc_windows first");
    const hit = lastWindows.items.find((x) => x.n === k);
    if (!hit) throw fail(`the last window list has only ${lastWindows.items.length} window(s)`);
    return hit;
  }

  async function executeLook(target, how, { confirmedSite = null } = {}) {
    pruneHour();
    if (hourLooks.length >= MAX_LOOKS_PER_HOUR) {
      throw fail(`Bot looks at most ${MAX_LOOKS_PER_HOUR} times per hour; wait a little before looking again`);
    }
    if (target.kind === "tab") {
      const never = whyNeverTab(target.url, target.title);
      if (never) throw fail(never);
      if (!bridge) throw fail("Chrome tab access is not set up in this Bot", { code: "offline" });
      if (how === "picture") {
        if (target.minimized) throw fail(`the Chrome window for "${oneLine(target.title, 70)}" is minimized, I cannot see it`);
        if (target.active === false) throw fail(`the "${oneLine(target.title, 70)}" tab is not in front in its Chrome window, I cannot take a picture of it (ask for its text instead, or bring the tab to the front yourself)`);
        let res;
        try {
          res = await bridge.ask("tab_shot", { tabId: target.id });
        } catch (error) {
          if (error.code === "minimized") throw fail(`the Chrome window for "${oneLine(target.title, 70)}" is minimized, I cannot see it`);
          if (error.code === "notfront") throw fail(`the "${oneLine(target.title, 70)}" tab is not in front in its Chrome window, I cannot take a picture of it`);
          throw error;
        }
        const liveUrl = oneLine((res.data && (res.data.url || res.data.address)) || target.url, 600);
        const liveTitle = oneLine((res.data && res.data.title) || target.title, 240);
        const liveNever = whyNeverTab(liveUrl, liveTitle);
        if (liveNever) throw fail(liveNever);
        const liveSite = siteOfUrl(liveUrl);
        if (!liveSite || (liveSite !== target.site && !isSiteAllowed(liveSite) && liveSite !== confirmedSite)) {
          throw fail(`that tab moved to another site (${liveSite || "unknown"}); call chrome_tabs again first`);
        }
        const rawBytes = res.file && res.file.bytes ? res.file.bytes : res.data && typeof res.data.base64 === "string" ? Buffer.from(res.data.base64, "base64") : null;
        const checked = checkLookImage(rawBytes);
        const saved = saveLookPhoto(checked);
        hourLooks.push(now());
        log(`[pc-look] looked at Chrome tab (picture): ${liveSite}`);
        return { status: "looked", kind: "tab", how: "picture", site: liveSite, title: liveTitle, label: `your ${liveSite} tab`, photoPath: saved.path, ext: saved.ext };
      }

      const res = await bridge.ask("tab_text", { tabId: target.id });
      const data = (res && res.data) || {};
      const liveUrl = oneLine(data.url || data.address || target.url, 600);
      const liveTitle = oneLine(data.title || target.title, 240);
      const liveNever = whyNeverTab(liveUrl, liveTitle);
      if (liveNever) throw fail(liveNever);
      const liveSite = siteOfUrl(liveUrl);
      if (!liveSite || (liveSite !== target.site && !isSiteAllowed(liveSite) && liveSite !== confirmedSite)) {
        throw fail(`that tab moved to another site (${liveSite || "unknown"}); call chrome_tabs again first`);
      }
      const text = String(data.text == null ? "" : data.text).slice(0, MAX_TAB_TEXT_CHARS);
      hourLooks.push(now());
      log(`[pc-look] looked at Chrome tab (text): ${liveSite}`);
      return {
        status: "looked",
        kind: "tab",
        how: "text",
        site: liveSite,
        title: liveTitle,
        url: liveUrl,
        label: `your ${liveSite} tab`,
        text: `Tab title: ${liveTitle}\nAddress: ${liveUrl}\n\n${text || "(the page has no visible text)"}`,
      };
    }

    // kind === "window"
    const never = whyNeverWindow(target.app, target.title);
    if (never) throw fail(never);
    const cap = await getWindowShot(target.hwnd);
    const name = shortApp(target.app);
    if (!cap || cap.status === "minimized") throw fail(`the ${name} window is minimized, I cannot see it`);
    if (cap.status === "closed" || cap.status === "hidden") throw fail(`the ${name} window is closed or hidden, I cannot see it`);
    if (cap.status !== "ok") throw fail(`Bot could not capture the ${name} window`);
    const liveTitle = oneLine(cap.title || target.title, 240);
    const liveNever = whyNeverWindow(target.app, liveTitle);
    if (liveNever) throw fail(liveNever);
    const checked = checkLookImage(cap.bytes);
    const saved = saveLookPhoto(checked);
    hourLooks.push(now());
    log(`[pc-look] looked at window (picture): ${target.app}`);
    return { status: "looked", kind: "window", how: "picture", app: target.app, title: liveTitle, label: `your ${name} window`, photoPath: saved.path, ext: saved.ext };
  }

  async function look({ kind, number, how, task = "", questionId = null, fromPhone = false, lang = "en" } = {}) {
    sweepCards();
    pruneHour();
    if (kind !== "tab" && kind !== "window") throw fail('say "kind":"tab" (a Chrome tab) or "kind":"window" (an open app window)');
    const mode = how === undefined || how === null || how === "" ? (kind === "window" ? "picture" : "text") : String(how).toLowerCase();
    if (mode !== "text" && mode !== "picture") throw fail('say "how":"text" or "how":"picture"');
    if (kind === "window" && mode === "text") throw fail('an app window can only be looked at as a picture ("how":"picture"), not text');
    if (hourLooks.length >= MAX_LOOKS_PER_HOUR) {
      throw fail(`Bot looks at most ${MAX_LOOKS_PER_HOUR} times per hour; wait a little before looking again`);
    }

    if (kind === "tab") {
      const tab = pickTab(number);
      const never = whyNeverTab(tab.url, tab.title);
      if (never) throw fail(never);
      if (!tab.site) throw fail("that tab does not have a standard web site address");
      if (mode === "picture") {
        if (tab.minimized) throw fail(`the Chrome window for "${oneLine(tab.title, 70)}" is minimized, I cannot see it`);
        if (tab.active === false) throw fail(`the "${oneLine(tab.title, 70)}" tab is not in front in its Chrome window, I cannot take a picture of it (use "how":"text" instead, or ask Omri to bring the tab to the front)`);
      }
      const target = { kind: "tab", ...tab };
      if (isSiteAllowed(tab.site)) return executeLook(target, mode);

      const id = crypto.randomBytes(8).toString("hex");
      const card = {
        id,
        kind: "tab",
        how: mode,
        site: tab.site,
        title: oneLine(tab.title, 140),
        expiresAt: now() + CARD_EXPIRES_MS,
      };
      pending.set(id, { id, at: now(), target, how: mode, task, questionId, fromPhone: Boolean(fromPhone), lang, card });
      while (pending.size > MAX_PENDING_CARDS) pending.delete(pending.keys().next().value);
      return {
        status: "confirm",
        card,
        text: `Site "${tab.site}" is not on Omri's allowed sites list yet. A Confirm card is now shown to Omri with three choices: "Just this once", "Always allow (add to the list)" and "Cancel". Bot has NOT looked at the tab yet — tell Omri in one short sentence that the card waits for their tap, and do not guess what is on the tab.`,
      };
    }

    const win = pickWindow(number);
    const never = whyNeverWindow(win.app, win.title);
    if (never) throw fail(never);
    if (win.minimized) throw fail(`the ${shortApp(win.app)} window is minimized, I cannot see it`);
    const target = { kind: "window", ...win };
    if (isAppAllowed(win.app)) return executeLook(target, "picture");

    const id = crypto.randomBytes(8).toString("hex");
    const card = {
      id,
      kind: "window",
      how: "picture",
      app: win.app,
      title: oneLine(win.title, 140),
      expiresAt: now() + CARD_EXPIRES_MS,
    };
    pending.set(id, { id, at: now(), target, how: "picture", task, questionId, fromPhone: Boolean(fromPhone), lang, card });
    while (pending.size > MAX_PENDING_CARDS) pending.delete(pending.keys().next().value);
    return {
      status: "confirm",
      card,
      text: `App "${win.app}" is not on Omri's allowed apps list yet. A Confirm card is now shown to Omri with three choices: "Just this once", "Always allow (add to the list)" and "Cancel". Bot has NOT looked at the window yet — tell Omri in one short sentence that the card waits for their tap, and do not guess what is in the window.`,
    };
  }

  function cancelCard(id) {
    sweepCards();
    if (!pending.delete(String(id || ""))) throw Object.assign(new Error("that card is no longer waiting (15 minutes pass, or it was already used)"), { status: 404 });
    return { ok: true, choice: "cancel", cancelled: true };
  }

  async function decide(id, choice) {
    sweepCards();
    if (!["once", "always", "cancel"].includes(choice)) throw Object.assign(new Error('choice must be "once", "always" or "cancel"'), { status: 400 });
    if (choice === "cancel") return cancelCard(id);
    const p = pending.get(String(id || ""));
    if (!p) throw Object.assign(new Error("that card is no longer waiting (15 minutes pass, or it was already used); ask again"), { status: 404 });
    pending.delete(p.id); // single-use: consumed before any work runs

    if (p.target.kind === "tab") {
      const never = whyNeverTab(p.target.url, p.target.title);
      if (never) throw Object.assign(fail(never), { status: 400 });
      if (choice === "always") allow({ site: p.target.site, fromUi: true });
      const lookRes = await executeLook(p.target, p.how, { confirmedSite: p.target.site });
      return { ok: true, choice, added: choice === "always", lists: lists(), card: p, look: lookRes };
    }

    const never = whyNeverWindow(p.target.app, p.target.title);
    if (never) throw Object.assign(fail(never), { status: 400 });
    if (choice === "always") allow({ app: p.target.app, fromUi: true });
    const lookRes = await executeLook(p.target, p.how);
    return { ok: true, choice, added: choice === "always", lists: lists(), card: p, look: lookRes };
  }

  return {
    lists,
    listsText,
    allow,
    readTabs,
    readWindows,
    pickTab,
    pickWindow,
    look,
    decide,
    cancelCard,
    isSiteAllowed,
    isAppAllowed,
  };
}
