# PC Look: Chrome Extension Changes (`moodle-extension/`)

Because `moodle-extension/` is a guard folder in Improve Bot, the server and brain side (`moodle-bridge.js`, `pc-look.js`, `brain.js`, `server.js`, `public/index.html`) are already wired and ready, and the exact extension files to copy into `moodle-extension/` are below.

## How Omri applies this once by hand

1. Replace `moodle-extension/manifest.json` and `moodle-extension/background.js` with the two files below.
2. Open `chrome://extensions` in Chrome.
3. Click **Reload** on **Bot Moodle reader**, and accept the updated permissions (`tabs` and `scripting`, plus `http://*/*` and `https://*/*` so Chrome lets the extension read page text or capture a visible tab when Bot asks; `chrome://`, `edge://`, `about:` and `file://` are never requested and are also refused in code).

Nothing in the extension clicks, types, focuses windows, switches tabs, closes tabs or submits anything. All existing Moodle read-only checks and exports (`run`, `JOBS`, `READ_METHODS`, `folderFiles`, `isFileUrl`, `pollOnce`, `_reset`) stay intact.

---

## `moodle-extension/manifest.json`

```json
{
  "manifest_version": 3,
  "name": "Bot Moodle reader",
  "version": "1.1",
  "description": "Lets Bot (on this PC) READ Omri's BGU Moodle and, when asked, list or look at open Chrome tabs. Read-only.",
  "permissions": ["alarms", "tabs", "scripting"],
  "host_permissions": [
    "https://moodle.bgu.ac.il/*",
    "http://localhost:3000/*",
    "http://localhost:3100/*",
    "https://*/*",
    "http://*/*"
  ],
  "background": { "service_worker": "background.js" }
}
```

## `moodle-extension/background.js`

```js
// Bot Moodle reader + PC Look Chrome bridge (loaded unpacked from chrome://extensions).
//
// READ-ONLY, enforced here and in Bot's Node code (moodle-bridge.js + pc-look.js):
//   - Bot sends only a job KIND and whole NUMBERS (never a URL or script).
//   - Moodle jobs: only GET of fixed Moodle pages or pluginfile.php addresses, and POST /lib/ajax/service.php with READ_METHODS.
//   - Tab jobs (2026-11-24):
//       tabs      -> lists id, title, url, active and minimized state of open Chrome tabs (chrome.tabs.query).
//       tab_text  -> reads plain document.body.innerText of one tab (only http/https, never internal/file/password/banking pages).
//       tab_shot  -> captures a PNG of one tab ONLY when its window is not minimized and the tab is already active in front.
//   - Never switches tabs, focuses windows, clicks, types, fills forms or changes anything.

const MOODLE = "https://moodle.bgu.ac.il/moodle";
const BOTS = ["http://localhost:3000", "http://localhost:3100"]; // 3100 = a test copy of Bot
const READ_METHODS = new Set([
  "core_course_get_enrolled_courses_by_timeline_classification",
  "core_courseformat_get_state",
  "core_calendar_get_action_events_by_timesort",
  "mod_forum_get_discussion_posts",
]);
const MAX_FILE_BYTES = 50 * 1024 * 1024;
const MAX_SHOT_BYTES = 8 * 1024 * 1024;
const MAX_TAB_TEXT = 60000;
const POLL_MS = 2000;
const HDR = { "X-Bot-Bridge": "1" };

// Defense-in-depth inside the extension (pc-look.js enforces the full NEVER_ALLOWED list on the Node side too).
const NEVER_SCHEMES = new Set(["chrome:", "edge:", "about:", "file:", "chrome-extension:", "devtools:", "view-source:", "javascript:", "data:"]);
const NEVER_HOST = /(^|[.-])(1password|bitwarden|keepass\w*|lastpass|dashlane|nordpass|enpass|roboform|passwords\.google|netbank\w*|onlinebanking|hapoalim|leumi|mizrahi|discountbank|isracard|paypal|wise|revolut|chase|bankofamerica|wellsfargo|citibank)(?=[.-]|$)|(^|[.-])bank(?:ing)?(?=[.-]|$)|^bank[a-z0-9-]*\.|[a-z0-9-]+bank\.(?:com|co\.il|org|net|eu|co\.uk)$/i;

const fail = (code, message) => Object.assign(new Error(message), { code });
let sesskey = null;
let loggedIn = null;

function checkTabUrl(raw) {
  let u;
  try { u = new URL(String(raw || "")); } catch (error) { throw fail("refused", "that tab has no standard web address"); }
  if (NEVER_SCHEMES.has(u.protocol.toLowerCase()) || (u.protocol !== "https:" && u.protocol !== "http:")) {
    throw fail("refused", `Bot never looks at ${u.protocol} pages`);
  }
  if (NEVER_HOST.test(u.hostname)) {
    throw fail("refused", `Bot never looks at password manager or banking sites (${u.hostname})`);
  }
  return u;
}

// ---- Moodle, read-only ----------------------------------------------------------------------------------------------------------
const isLogin = (url) => /\/login\/index\.php/.test(new URL(url).pathname);

async function session(force) {
  if (sesskey && !force) return sesskey;
  const r = await fetch(`${MOODLE}/my/`, { credentials: "include" });
  const html = await r.text();
  const m = /"sesskey"\s*:\s*"([A-Za-z0-9]{5,40})"/.exec(html);
  loggedIn = !!m && !isLogin(r.url) && /login\/logout\.php/.test(html);
  sesskey = loggedIn ? m[1] : null;
  if (!sesskey) throw fail("loggedout", "not logged in to Moodle in Chrome");
  return sesskey;
}

async function ajax(methodname, args) {
  if (!READ_METHODS.has(methodname)) throw fail("refused", `${methodname} is not on the read-only list`);
  for (let attempt = 0; attempt < 2; attempt++) {
    const key = await session(attempt > 0);
    const r = await fetch(`${MOODLE}/lib/ajax/service.php?sesskey=${encodeURIComponent(key)}&info=${methodname}`, {
      method: "POST", credentials: "include", headers: { "Content-Type": "application/json" },
      body: JSON.stringify([{ index: 0, methodname, args }]),
    });
    let j = null;
    try { j = await r.json(); } catch (error) { /* checked below */ }
    const d = Array.isArray(j) ? j[0] : j;
    if (d && !d.error && Array.isArray(j)) return d.data;
    const code = String((d && (d.errorcode || (d.exception && d.exception.errorcode))) || "");
    if (/servicerequireslogin|invalidsesskey|requirelogin/.test(code)) { sesskey = null; if (attempt === 0) continue; throw fail("loggedout", "not logged in to Moodle in Chrome"); }
    throw fail("moodle", String((d && ((d.exception && d.exception.message) || d.message)) || code || "Moodle refused").slice(0, 200));
  }
}

async function page(path, id) {
  if (!/^\/mod\/(forum|resource|folder)\/view\.php$/.test(path)) throw fail("refused", "not a page Bot may open");
  const r = await fetch(`${MOODLE}${path}?id=${Number(id)}`, { credentials: "include" });
  if (isLogin(r.url)) { loggedIn = false; sesskey = null; throw fail("loggedout", "not logged in to Moodle in Chrome"); }
  if (!r.ok) throw fail("moodle", `Moodle answered ${r.status}`);
  return { r, url: r.url };
}

const decode = (s) => String(s).replace(/&amp;/g, "&").replace(/&quot;/g, "\"").replace(/&#0?39;/g, "'").replace(/&lt;/g, "<").replace(/&gt;/g, ">");
const plain = (s) => decode(String(s).replace(/<[^>]+>/g, " ")).replace(/\s+/g, " ").trim();
const FILE_ORIGIN = new URL(MOODLE).origin, FILE_PATH = `${new URL(MOODLE).pathname}/pluginfile.php/`;
function isFileUrl(u) {
  if (typeof u !== "string" || /[\u0000-\u001f\\]/.test(u)) return false;
  let url;
  try { url = new URL(u); } catch (error) { return false; }
  let query = url.search;
  try { query = decodeURIComponent(query); } catch (error) { /* judged as written */ }
  return url.origin === FILE_ORIGIN && !url.username && !url.password && url.pathname.startsWith(FILE_PATH) && !/sesskey\s*=/i.test(query);
}
const fileName = (u) => { const last = new URL(u).pathname.split("/").pop() || "file"; try { return decodeURIComponent(last); } catch (error) { return last; } };

async function download(url, response) {
  if (!isFileUrl(url)) throw fail("refused", "only Moodle file addresses are downloaded");
  const r = response || await fetch(url, { credentials: "include" });
  if (isLogin(r.url)) throw fail("loggedout", "not logged in to Moodle in Chrome");
  if (!r.ok) throw fail("moodle", `Moodle answered ${r.status}`);
  const declared = Number(r.headers.get("content-length"));
  if (declared > MAX_FILE_BYTES) throw fail("toobig", "the file is bigger than 50 MB");
  const buf = new Uint8Array(await r.arrayBuffer());
  if (buf.length > MAX_FILE_BYTES) throw fail("toobig", "the file is bigger than 50 MB");
  let bin = "";
  for (let i = 0; i < buf.length; i += 0x8000) bin += String.fromCharCode.apply(null, buf.subarray(i, i + 0x8000));
  return { name: fileName(r.url || url), type: r.headers.get("content-type") || "", base64: btoa(bin) };
}

function folderFiles(html) {
  const seen = new Set();
  const out = [];
  for (const m of html.matchAll(/href="([^"]*\/pluginfile\.php\/\d+\/mod_folder\/content\/[^"]+)"/g)) {
    const u = decode(m[1]).replace(/[?&]forcedownload=1$/, "");
    if (!isFileUrl(u) || seen.has(u)) continue;
    seen.add(u);
    out.push(u);
  }
  return out;
}

const JOBS = {
  async courses() {
    const d = await ajax("core_course_get_enrolled_courses_by_timeline_classification", { offset: 0, limit: 0, classification: "inprogress", sort: "fullname" });
    return { data: (d.courses || []).map((c) => ({ id: c.id, name: c.fullname, short: c.shortname, category: c.coursecategory, start: c.startdate })) };
  },
  async course({ courseid }) {
    const raw = await ajax("core_courseformat_get_state", { courseid });
    const st = typeof raw === "string" ? JSON.parse(raw) : raw;
    return { data: {
      name: st.course && st.course.fullname || null,
      sections: (st.section || []).map((s) => ({ id: s.id, number: s.number, title: s.title, cms: s.cmlist || [] })),
      items: (st.cm || []).filter((c) => c.uservisible !== false).map((c) => ({ id: Number(c.id), name: c.name, module: c.module, section: c.sectionid })),
    } };
  },
  async forum({ cmid }) {
    const { r } = await page("/mod/forum/view.php", cmid);
    const html = await r.text();
    const rows = html.split(/data-region="discussion-list-item"/).slice(1);
    const list = [];
    for (const row of rows) {
      const a = /href="[^"]*\/mod\/forum\/discuss\.php\?d=(\d+)[^"]*"[^>]*>([\s\S]*?)<\/a>/.exec(row);
      if (!a) continue;
      const t = /data-timestamp="(\d+)"/.exec(row);
      list.push({ d: Number(a[1]), title: plain(a[2]).slice(0, 300), time: t ? Number(t[1]) : null });
    }
    return { data: list.slice(0, 50) };
  },
  async posts({ discussionid }) {
    const d = await ajax("mod_forum_get_discussion_posts", { discussionid, sortby: "created", sortdirection: "ASC", includeinlineattachments: false });
    return { data: (d.posts || []).slice(0, 20).map((p) => ({ subject: p.subject, author: p.author && p.author.fullname || "", created: p.timecreated, message: String(p.message || "").slice(0, 60000), attachments: (p.attachments || []).map((x) => x.filename) })) };
  },
  async events() {
    const d = await ajax("core_calendar_get_action_events_by_timesort", { timesortfrom: Math.floor(Date.now() / 1000) - 86400, limitnum: 30 });
    return { data: (d.events || []).map((e) => ({ name: e.name, courseId: e.course && e.course.id, course: e.course && e.course.fullname, due: e.timesort, module: e.modulename })) };
  },
  async file({ cmid }) {
    const { r, url } = await page("/mod/resource/view.php", cmid);
    if (isFileUrl(url)) return { file: await download(url, r) };
    const html = await r.text();
    const m = /(https:\/\/moodle\.bgu\.ac\.il\/moodle\/pluginfile\.php\/\d+\/mod_resource\/content\/[^"'<>\s]+)/.exec(html);
    if (!m) throw fail("notfile", "this item has no file to download");
    return { file: await download(decode(m[1]).replace(/[?&]forcedownload=1$/, "")) };
  },
  async folder({ cmid }) {
    const { r } = await page("/mod/folder/view.php", cmid);
    return { data: folderFiles(await r.text()).slice(0, 300).map((u, index) => ({ index, name: fileName(u) })) };
  },
  async folderfile({ cmid, index }) {
    const { r } = await page("/mod/folder/view.php", cmid);
    const u = folderFiles(await r.text())[index];
    if (!u) throw fail("notfile", "that file is not in the folder");
    return { file: await download(u) };
  },
  // ---- PC Look: read-only Chrome tabs (never focuses, switches, clicks or types) ------------------------------------------------
  async tabs() {
    if (typeof chrome === "undefined" || !chrome.tabs) throw fail("refused", "Chrome tabs permission is not active yet (reload the extension in chrome://extensions)");
    const [allTabs, allWins] = await Promise.all([chrome.tabs.query({}), chrome.windows ? chrome.windows.getAll() : []]);
    const winState = new Map((allWins || []).map((w) => [w.id, w.state]));
    return {
      data: (allTabs || []).slice(0, 200).map((t) => ({
        id: Number(t.id),
        title: String(t.title || "").slice(0, 300),
        url: String(t.url || "").slice(0, 1000),
        active: Boolean(t.active),
        minimized: winState.get(t.windowId) === "minimized",
      })),
    };
  },
  async tab_text({ tabId }) {
    if (typeof chrome === "undefined" || !chrome.tabs || !chrome.scripting) throw fail("refused", "Chrome tabs/scripting permission is not active yet (reload the extension)");
    let tab;
    try { tab = await chrome.tabs.get(Number(tabId)); } catch (error) { throw fail("notfound", "that Chrome tab is no longer open"); }
    checkTabUrl(tab.url);
    const results = await chrome.scripting.executeScript({
      target: { tabId: Number(tabId) },
      func: () => ({
        title: String(document.title || "").slice(0, 300),
        url: String(location.href || "").slice(0, 1000),
        text: String((document.body && document.body.innerText) || "").slice(0, 60000),
      }),
    });
    const got = results && results[0] && results[0].result ? results[0].result : { title: tab.title || "", url: tab.url || "", text: "" };
    checkTabUrl(got.url || tab.url);
    return { data: got };
  },
  async tab_shot({ tabId }) {
    if (typeof chrome === "undefined" || !chrome.tabs || !chrome.windows) throw fail("refused", "Chrome tabs permission is not active yet (reload the extension)");
    let tab;
    try { tab = await chrome.tabs.get(Number(tabId)); } catch (error) { throw fail("notfound", "that Chrome tab is no longer open"); }
    checkTabUrl(tab.url);
    const win = await chrome.windows.get(tab.windowId);
    if (win && win.state === "minimized") throw fail("minimized", "that Chrome window is minimized, I cannot see it");
    if (!tab.active) throw fail("notfront", "that tab is not in front in its Chrome window, I cannot take a picture of it");
    let dataUrl = await chrome.tabs.captureVisibleTab(tab.windowId, { format: "png" });
    let mime = "image/png", name = "tab.png";
    let b64 = String(dataUrl || "").replace(/^data:image\/png;base64,/, "");
    if (Math.floor((b64.length * 3) / 4) > MAX_SHOT_BYTES) {
      dataUrl = await chrome.tabs.captureVisibleTab(tab.windowId, { format: "jpeg", quality: 82 });
      mime = "image/jpeg";
      name = "tab.jpg";
      b64 = String(dataUrl || "").replace(/^data:image\/jpeg;base64,/, "");
    }
    if (Math.floor((b64.length * 3) / 4) > MAX_SHOT_BYTES) throw fail("toobig", "the screenshot is bigger than 8 MB");
    return { data: { title: String(tab.title || "").slice(0, 300), url: String(tab.url || "").slice(0, 1000) }, file: { name, type: mime, base64: b64 } };
  },
};

async function run(job) {
  const fn = Object.prototype.hasOwnProperty.call(JOBS, job.kind) ? JOBS[job.kind] : null;
  if (!fn) throw fail("refused", "unknown job");
  return fn(job.args || {});
}

// ---- talking to Bot ---------------------------------------------------------------------------------------------------------------
async function pollOnce(bot) {
  let job;
  try {
    const r = await fetch(`${bot}/api/moodle/next?loggedIn=${loggedIn === null ? "" : loggedIn ? 1 : 0}`, { headers: HDR });
    if (!r.ok) return false;
    job = (await r.json()).job;
  } catch (error) { return false; }
  if (!job) return false;
  let body;
  try { body = { id: job.id, ok: true, ...(await run(job)) }; } catch (error) { body = { id: job.id, ok: false, code: error.code || "moodle", error: String(error.message || error).slice(0, 300) }; }
  try { await fetch(`${bot}/api/moodle/result`, { method: "POST", headers: { ...HDR, "Content-Type": "application/json" }, body: JSON.stringify(body) }); } catch (error) { /* Bot went away */ }
  return true;
}

let running = false;
async function loop() {
  if (running) return;
  running = true;
  try {
    for (;;) {
      let busy = false;
      for (const bot of BOTS) busy = (await pollOnce(bot)) || busy;
      if (typeof chrome !== "undefined" && chrome.runtime && chrome.runtime.getPlatformInfo) await chrome.runtime.getPlatformInfo();
      if (!busy) await new Promise((r) => setTimeout(r, POLL_MS));
    }
  } finally { running = false; }
}

if (typeof chrome !== "undefined" && chrome.alarms) {
  chrome.alarms.create("bot-moodle", { periodInMinutes: 0.5 });
  chrome.alarms.onAlarm.addListener(() => loop());
  chrome.runtime.onStartup.addListener(() => loop());
  chrome.runtime.onInstalled.addListener(() => loop());
  loop();
}
if (typeof module !== "undefined") module.exports = { run, JOBS, READ_METHODS, folderFiles, isFileUrl, checkTabUrl, pollOnce, _reset: () => { sesskey = null; loggedIn = null; } };
```
